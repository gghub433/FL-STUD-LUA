# Native parts of the FL LUA desktop app:
#   fllua-plughost     the VST3 / CLAP plugin host (a process of its own)
#   fllua_bridge.node  the page-process bridge (a Node-API module, see bridge/bridge.cc)
#   test plugins       a tiny VST3 and CLAP synth + effect used by the tests (not shipped)
cmake_minimum_required(VERSION 3.16)
project(fllua_native CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "" FORCE)
endif()
if(MSVC)
  add_compile_options(/utf-8 /W3 /EHsc)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX UNICODE _UNICODE)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(VST3 ${CMAKE_CURRENT_SOURCE_DIR}/third_party/vst3)
set(CLAP ${CMAKE_CURRENT_SOURCE_DIR}/third_party/clap/include)
set(VST3_BASE ${VST3}/pluginterfaces/base/funknown.cpp)  # coreiids.cpp is not used: the *_iids.cpp files define every id they include

# ---------------------------------------------------------------- host
set(HOST_SRC host/main.cpp host/vst3.cpp host/vst3_iids.cpp host/clap.cpp ${VST3_BASE})
if(WIN32)
  list(APPEND HOST_SRC host/platform_win.cpp)
elseif(APPLE)
  list(APPEND HOST_SRC host/platform_mac.mm)
else()
  list(APPEND HOST_SRC host/platform_x11.cpp)
endif()
add_executable(fllua-plughost ${HOST_SRC})
target_include_directories(fllua-plughost PRIVATE ${VST3} ${CLAP} host common)
if(WIN32)
  target_link_libraries(fllua-plughost PRIVATE avrt user32 gdi32 ole32)
  target_compile_definitions(fllua-plughost PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
elseif(APPLE)
  target_link_libraries(fllua-plughost PRIVATE "-framework Cocoa" "-framework CoreFoundation")
  set_source_files_properties(host/platform_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
else()
  find_package(X11 REQUIRED)
  target_link_libraries(fllua-plughost PRIVATE ${X11_LIBRARIES} pthread dl rt)
endif()

# ---------------------------------------------------------------- bridge (Node-API module; no node-gyp, no node.lib)
add_library(fllua-bridge MODULE bridge/bridge.cc)
set_target_properties(fllua-bridge PROPERTIES PREFIX "" OUTPUT_NAME fllua_bridge SUFFIX ".node")
target_include_directories(fllua-bridge PRIVATE third_party/node-api ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_sources(fllua-bridge PRIVATE bridge/napi_win.cc)           # finds the Node-API functions in the loading exe
  target_compile_definitions(fllua-bridge PRIVATE NAPI_EXTERN= _WIN32_WINNT=0x0A00 WINVER=0x0A00)
elseif(APPLE)
  target_link_options(fllua-bridge PRIVATE -undefined dynamic_lookup)
else()
  target_link_libraries(fllua-bridge PRIVATE pthread rt)
endif()

# ---------------------------------------------------------------- test plugins
option(FLLUA_TEST_PLUGINS "Build the test plugins" ON)
if(FLLUA_TEST_PLUGINS)
  add_library(fllua-test-vst3 MODULE testplugins/vst3_test.cpp testplugins/vst3_test_iids.cpp ${VST3_BASE})
  target_include_directories(fllua-test-vst3 PRIVATE ${VST3})
  set_target_properties(fllua-test-vst3 PROPERTIES PREFIX "" OUTPUT_NAME "FLLUA Test VST3")
  add_library(fllua-test-clap MODULE testplugins/clap_test.cpp)
  target_include_directories(fllua-test-clap PRIVATE ${CLAP})
  set_target_properties(fllua-test-clap PROPERTIES PREFIX "" OUTPUT_NAME "FLLUA Test CLAP")
  if(APPLE)
    set_target_properties(fllua-test-vst3 PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION vst3)
    set_target_properties(fllua-test-clap PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION clap)
  elseif(UNIX)
    target_link_libraries(fllua-test-vst3 PRIVATE X11)
    target_link_libraries(fllua-test-clap PRIVATE X11)
  else()
    target_link_libraries(fllua-test-vst3 PRIVATE user32 gdi32)
    target_link_libraries(fllua-test-clap PRIVATE user32 gdi32)
  endif()
  if(NOT APPLE)
    set_target_properties(fllua-test-clap PROPERTIES SUFFIX ".clap")
  endif()
endif()
